# set_camera.txt — Set the viewport view mode and position the editor camera.
#
# Sample for the viewport skill. It is a MIXED recipe: the view mode and the camera transform are
# Epic's toolsets (call_tool), the FOV and the read-back are VibeUE's ViewportService
# (execute_python). Epic's toolsets carry no PythonCallable, so unreal.EditorAppToolset does not
# exist — do not try to call them from Python.

# 1) View mode — Epic's EditorAppToolset (VibeUE's set_view_mode was removed).
#    Takes the full EViewModeIndex; describe_toolset EditorAppToolset lists the live names.
call_tool SetViewportViewMode {"ViewMode": "VMI_Lit"}

# 2) Camera transform — Epic's EditorAppToolset (FocusOnActors frames actors instead of guessing).
call_tool SetCameraTransform {"Location": {"X": 1000, "Y": 1000, "Z": 800},
                              "Rotation": {"Pitch": -30, "Yaw": -135, "Roll": 0}}

# 3) FOV — still VibeUE's, because Epic ships no FOV writer.
#    WARNING: set_fov persists into EditorPerProjectUserSettings.ini. Snapshot and restore.
execute_python
import unreal
vp = unreal.ViewportService

info = vp.get_viewport_info()          # snapshot FIRST (camera, FOV, exposure, realtime, game view)
try:
    vp.set_viewport_type("perspective")  # FOV only applies in perspective
    vp.set_fov(75.0)
    # ... do the work that needs this view ...
    print("view mode:", vp.get_view_mode())   # the reader is VibeUE's; the setter is Epic's
    print("viewport:", vp.get_viewport_info())
finally:
    vp.restore_viewport_info(info)      # put the user's FOV/exposure/camera back

# Read the type/FOV from Epic's side instead if you prefer (get_viewport_type / get_fov were
# removed): call_tool GetActiveEditorView {} -> Camera.FOV, Camera.ProjectionMode;
# call_tool ListEditorViewports {} -> one entry per client, DisplayName carries its viewport type.
