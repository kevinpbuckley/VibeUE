# timer_with_custom_event.txt — Non-blocking timer: Set Timer by Event + a custom-event callback,
# authored as one Graph DSL document.
#
# Sample script for the blueprint-graphs skill. Run via execute_python.
#
# Use this pattern (NOT Delay) when the Blueprint must keep ticking during the wait:
#   Delay               is a latent action and BLOCKS that execution path.
#   Set Timer by Event  is non-blocking and fires a delegate callback after the duration.
# For a State Tree task that rotates while waiting, an animation task, or anything that must keep
# running, Set Timer by Event is the only correct choice.
#
# UE 5.9 notes:
#   - add_custom_event_node / connect_nodes / set_node_pin_value / compile_blueprint were all cut
#     from BlueprintService. A custom event is now an (event `Name` ...) entry in a DSL document,
#     and the delegate wiring is (`|CreateDelegate` :!event `Name`).
#   - BlueprintDslToolset is AICallable only — reach it with vibeue.exec_tool (or call_tool from
#     an MCP client), never as unreal.BlueprintDslToolset.
import unreal, vibeue

DSL = "BlueprintToolsets.BlueprintDslToolset"
BT  = "BlueprintToolsets.BlueprintToolset"

BP    = "/Game/Blueprints/BP_GraphTest"      # must already exist
BPP   = BP + ".BP_GraphTest"
GRAPH = "EventGraph"


def ref(p):
    tail = p.rsplit("/", 1)[-1]
    return {"refPath": p if "." in tail else "{}.{}".format(p, tail)}


les = unreal.get_editor_subsystem(unreal.LevelEditorSubsystem)
assert not les.is_in_play_in_editor(), "stop PIE before editing a Blueprint"

# 1. Grammar first — this document uses |CreateDelegate, :!event and keyword arguments, and
#    guessing any of those produces a document that will not parse.
vibeue.exec_tool(DSL, "Docs", {})

# 2. Confirm the timer function's pins before writing it. K2_SetTimerDelegate's inputs are
#    Delegate / Time / bLooping / bMaxOncePerFrame; keyword arguments are safer than positional.
print(vibeue.exec_tool(DSL, "Schema",
                       {"assetPath": BPP, "nodeType": "KismetSystemLibrary.K2_SetTimerDelegate"}))

# 3. Read the scope and keep the revision.
before = vibeue.exec_tool(DSL, "Read", {"assetPath": BPP, "scope": GRAPH})
rev = before["revision"]

# 4. One document, two entries: the caller and the callback. A Write creates the missing custom
#    event for us. Upsert means the Blueprint's other entries are left alone.
DOC = """
(event `self.ReceiveBeginPlay`
  (`KismetSystemLibrary.K2_SetTimerDelegate`
    :`Delegate` (`|CreateDelegate` :!event `OnTimerFinished`)
    :`Time` 1.0
    :`bLooping` false))

(event `OnTimerFinished`
  (`KismetSystemLibrary.PrintString` "timer fired"))
"""
res = vibeue.exec_tool(DSL, "Write", {"assetPath": BPP, "dsl": DOC, "scope": GRAPH})
rev = res["revision"]

# 5. Retune the delay with an Edit against that revision. On a stale token the call fails and the
#    asset is untouched: Read again and rebuild the replacement strings against the fresh text.
try:
    e = vibeue.exec_tool(DSL, "Edit", {"assetPath": BPP, "revision": rev, "scope": GRAPH,
                                       "edits": [{"oldString": ":`Time` 1.0",
                                                  "newString": ":`Time` 2.5"}]})
    rev = e["revision"]
except RuntimeError as err:
    print("edit refused (stale revision or non-unique OldString):", err)
    fresh = vibeue.exec_tool(DSL, "Read", {"assetPath": BPP, "scope": GRAPH})
    rev = fresh["revision"]

# 6. Verify the shape from the canonical text, not from what we sent. A Create Event /
#    CreateDelegate wrapper around a FUNCTION is a different graph shape from a real custom
#    event — the (event `OnTimerFinished` ...) entry below is the proof that we got the latter.
after = vibeue.exec_tool(DSL, "Read", {"assetPath": BPP, "scope": GRAPH})
print(after["dsl"])
assert "OnTimerFinished" in after["dsl"], "custom event entry missing after write"

vibeue.exec_tool(BT, "CompileBlueprint", {"blueprint": ref(BP)})
unreal.EditorAssetLibrary.save_asset(BP)
