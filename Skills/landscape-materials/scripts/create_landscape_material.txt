# SPLIT: compiling is Epic's MaterialToolset.Recompile (VibeUE's MaterialService is GONE) and is
# slow, so it is a SEPARATE call_tool step — never in this block. Always create LayerInfo objects
# for layers.
import unreal
lms = unreal.LandscapeMaterialService

# The material asset itself is Epic's MaterialToolset.CreateMaterial (returns a {refPath} handle);
# LandscapeMaterialService only adds the landscape nodes.
path = "/Game/Landscape/M_Landscape"
if not await call_tool("EditorToolset.AssetToolset.Exists", path=path):
    r = await call_tool("EditorToolset.MaterialToolset.CreateMaterial",
                        folderPath="/Game/Landscape", assetName="M_Landscape")
    print("CREATED:", r)

# Build a LayerBlend node with named layers (Grass, Rock), wire to BaseColor, etc.
# blend = lms.create_layer_blend_node_with_layers(path, ["Grass", "Rock"], -400, 0)
# ... connect textures per layer with lms.setup_layer_textures(...) (see workflows.md) ...

unreal.EditorAssetLibrary.save_asset(path)
print("done — now run the two engine steps, each as its own call_tool:")
print('  call_tool MaterialToolset.ConnectToOutput '
      '{"expression": {"refPath": "<the LandscapeLayerBlend from GetExpressions>"}, '
      '"outputName": "", "materialProperty": "MP_BaseColor"}')
print('  call_tool MaterialToolset.Recompile {"materialOrFunction": {"refPath": "%s.%s"}}'
      % (path, path.rsplit("/", 1)[-1]))
