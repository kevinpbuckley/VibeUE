# create_instance.txt — Create a material instance and override parameters.
#
# Sample script for the materials skill. Run via execute_python.
#
# UE 5.9: instances are Epic's EditorToolset.MaterialInstanceToolset. UMaterialService is DELETED,
# so set_instance_parameters_bulk / create_material_instance no longer exist.
#
# From an MCP client:
#   MIT = "EditorToolset.MaterialInstanceToolset"
#   call_tool(tool_name="Create", toolset_name=MIT,
#             arguments={"folderPath": FOLDER, "assetName": NAME, "parent": {"refPath": "..."}})
# Inside execute_python the same tools are reached with vibeue.exec_tool(toolset, tool, args).
#
# NOTE the arg order: folderPath, assetName, parent — NOT the old (parent, name, folder).
# Prefer MaterialInstanceToolset over MaterialEditingLibrary's setters: MEL's
# set_material_instance_scalar_parameter_value returns False on success, while these raise on a
# real failure (e.g. a name that is not a parameter of that type).
import unreal, vibeue

MIT = "EditorToolset.MaterialInstanceToolset"
AT  = "EditorToolset.AssetToolset"

PARENT = "/Game/Materials/M_Character"
NAME, FOLDER = "MI_PlayerRed", "/Game/Materials/"
ip = FOLDER + NAME


def ref(p):
    tail = p.rsplit("/", 1)[-1]
    return {"refPath": p if "." in tail else "{}.{}".format(p, tail)}


if not vibeue.exec_tool(AT, "Exists", {"path": PARENT}):
    raise RuntimeError("parent material missing: " + PARENT)

# What can be overridden, and with which setter. Entries are {"name": ..., "type": ...} with type
# Scalar | Vector | Texture | StaticSwitch. Setting a name of the wrong type raises.
params = vibeue.exec_tool(MIT, "ListParameters", {"material": ref(PARENT)})
print("parent parameters:", params)

if vibeue.exec_tool(AT, "Exists", {"path": ip}):
    vibeue.exec_tool(AT, "Delete", {"path": ip})

mi = vibeue.exec_tool(MIT, "Create", {"folderPath": FOLDER, "assetName": NAME,
                                      "parent": ref(PARENT)})

vibeue.exec_tool(MIT, "SetVectorParameter", {
    "instance": mi, "name": "BaseColor", "value": {"r": 1.0, "g": 0.0, "b": 0.0, "a": 1.0}})
vibeue.exec_tool(MIT, "SetScalarParameter", {"instance": mi, "name": "Roughness", "value": 0.3})

vibeue.exec_tool(AT, "SaveAsset", {"assetPath": ip})

# Getters return the EFFECTIVE value, inheriting from the parent when not overridden.
print("instance:", ip,
      "roughness:", vibeue.exec_tool(MIT, "GetScalarParameter", {"instance": mi, "name": "Roughness"}),
      "basecolor:", vibeue.exec_tool(MIT, "GetVectorParameter", {"instance": mi, "name": "BaseColor"}))

# GetStatistics takes a MaterialInterface, so an instance's compiled cost measures too.
print("stats:", vibeue.exec_tool("EditorToolset.MaterialToolset", "GetStatistics",
                                 {"material": ref(ip)}))
