# create_material.txt — Create a material with BaseColor + Roughness params, compile, verify.
#
# Sample script for the materials skill. Run via execute_python.
#
# UE 5.9: material authoring is Epic's EditorToolset.MaterialToolset, and every property goes
# through EditorToolset.ObjectToolset. UMaterialService is DELETED; unreal.MaterialNodeService
# keeps only create_function_call and create_collection_parameter (neither is needed here).
#
# From an MCP client you would call these with call_tool:
#   call_tool(tool_name="CreateMaterial", toolset_name="EditorToolset.MaterialToolset",
#             arguments={"folderPath": FOLDER, "assetName": NAME})
# Inside execute_python the same tools are reached with vibeue.exec_tool(toolset, tool, args),
# which unwraps "returnValue" and raises RuntimeError on a tool error. Object/class arguments are
# {"refPath": "/Game/Dir/Asset.Asset"} handles either way.
import json, unreal, vibeue

MT = "EditorToolset.MaterialToolset"
OT = "EditorToolset.ObjectToolset"
AT = "EditorToolset.AssetToolset"
E  = "/Script/Engine."

NAME, FOLDER = "M_Character", "/Game/Materials/"
path = FOLDER + NAME


def ref(p):
    tail = p.rsplit("/", 1)[-1]
    return {"refPath": p if "." in tail else "{}.{}".format(p, tail)}


def add(cls, x, y):
    return vibeue.exec_tool(MT, "AddExpression", {
        "materialOrFunction": ref(path), "expressionClass": {"refPath": E + cls},
        "x": x, "y": y})


def set_props(obj, values):
    return vibeue.exec_tool(OT, "SetProperties", {"instance": obj, "values": json.dumps(values)})


if vibeue.exec_tool(AT, "Exists", {"path": path}):
    vibeue.exec_tool(AT, "Delete", {"path": path})

# CreateMaterial returns the new Material as a {refPath} handle — interchangeable with ref(path),
# which the add()/set_props() helpers above use.
print("created:", vibeue.exec_tool(MT, "CreateMaterial",
                                   {"folderPath": FOLDER, "assetName": NAME}))

color = add("MaterialExpressionVectorParameter", -500, 0)
set_props(color, {"ParameterName": "BaseColor", "Group": "Surface",
                  "DefaultValue": {"r": 0.8, "g": 0.2, "b": 0.2, "a": 1.0}})
vibeue.exec_tool(MT, "ConnectToOutput", {"expression": color, "outputName": "",
                                         "materialProperty": "MP_BaseColor"})

rough = add("MaterialExpressionScalarParameter", -500, 150)
set_props(rough, {"ParameterName": "Roughness", "Group": "Surface", "DefaultValue": 0.5})
vibeue.exec_tool(MT, "ConnectToOutput", {"expression": rough, "outputName": "",
                                         "materialProperty": "MP_Roughness"})

# LayoutExpressions DOES exist in 5.9 — approximate x/y above, then auto-arrange.
vibeue.exec_tool(MT, "LayoutExpressions", {"materialOrFunction": ref(path)})

# Recompile RAISES with the failing node's object path on a compile error; silence means clean.
vibeue.exec_tool(MT, "Recompile", {"materialOrFunction": ref(path)})
vibeue.exec_tool(AT, "SaveAsset", {"assetPath": path})

stats = vibeue.exec_tool(MT, "GetStatistics", {"material": ref(path)})
print("compiled ok; pixel instructions:", stats["num_pixel_shader_instructions"],
      "samplers:", stats["num_samplers"],
      "pixel texture samples:", stats["num_pixel_texture_samples"])
print("groups:", vibeue.exec_tool(MT, "ListParameterGroups", {"materialOrFunction": ref(path)}))
