# safe_discovery.txt — Discover skeletons + animations WITHOUT loading assets (no crashes).
#
# Sample script for the animation-editing skill. Run via execute_python.
# NEVER unreal.load_asset() in a loop — it causes access violations. Use these metadata queries.
import unreal, vibeue
ass = unreal.AnimSequenceService

# Skeletons in the project. VibeUE's own skeleton list call is gone — this is Epic's
# EditorToolset.AssetToolset, which is a genuinely async tool: it is fired on one execute_python
# call and collected on the next, so run this script TWICE (the second run prints the results).
KEY = "EditorToolset.AssetToolset.FindAssets"
try:
    skeletons = vibeue.collect_tool_result(KEY)
except KeyError:
    skeletons = None
    vibeue.exec_tool_async(
        "EditorToolset.AssetToolset", "FindAssets",
        {"filter": {"assetType": {"refPath": "/Script/Engine.Skeleton"},
                    "folderPath": "/Game", "maxResults": 0}})

if skeletons is None:
    print("FindAssets fired - run this script again to collect the skeleton list")
else:
    print("skeletons:", len(skeletons))
    # Animations for the first skeleton (metadata only, no loading).
    # FindAssets returns OBJECT paths ("/Game/.../SK_X.SK_X") - strip the suffix for VibeUE.
    if skeletons:
        skeleton_path = skeletons[0].rsplit(".", 1)[0]
        for a in ass.find_animations_for_skeleton(skeleton_path)[:10]:
            print(f"{a.anim_name}  frames={a.frame_count}  dur={a.duration:.2f}s  {a.anim_path}")
