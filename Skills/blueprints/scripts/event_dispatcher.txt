# event_dispatcher.txt — Add a Blueprint Event Dispatcher (multicast delegate) with a parameter,
# broadcast it from the event graph, and subscribe to it from another Blueprint.
#
# Sample script for the blueprints skill. Run via execute_python.
#
# UE 5.9:
#   - The dispatcher itself is Epic's BlueprintToolset.AddEventDispatcher, which returns the
#     dispatcher's SIGNATURE GRAPH. Its parameters are AddFunctionParam on that graph (inputs
#     only). Removing one is RemoveFunctionGraph — there is no RemoveEventDispatcher.
#   - BROADCASTING is a Graph DSL statement now. add_call_delegate_node, add_event_dispatcher,
#     add_event_dispatcher_parameter and add_delegate_bind_* were all cut from BlueprintService.
#   - "EventDispatcher" is NOT a variable type string. Never try AddVariable for a dispatcher.
import unreal, vibeue

BT  = "BlueprintToolsets.BlueprintToolset"
DSL = "BlueprintToolsets.BlueprintDslToolset"
AS  = "EditorToolset.AssetToolset"

BP_PATH = "/Game/Blueprints/BP_Player"
BPP     = BP_PATH + ".BP_Player"          # DSL AssetPath: full object path


def ref(p):
    tail = p.rsplit("/", 1)[-1]
    return {"refPath": p if "." in tail else "{}.{}".format(p, tail)}


les = unreal.get_editor_subsystem(unreal.LevelEditorSubsystem)
assert not les.is_in_play_in_editor(), "stop PIE before editing a Blueprint"

bp = ref(BP_PATH)

# 1. The dispatcher. The return value is its signature graph — keep it.
sig = vibeue.exec_tool(BT, "AddEventDispatcher", {"blueprint": bp, "name": "OnDied"})
print("signature graph:", sig)

# 2. Its parameters, on that graph. Dispatchers take INPUTS ONLY (outputs are refused).
vibeue.exec_tool(BT, "AddFunctionParam", {"graph": sig, "paramName": "Killer",
                                          "paramType": "Actor", "inputParam": True})
vibeue.exec_tool(BT, "CompileBlueprint", {"blueprint": bp})
print("dispatchers:", vibeue.exec_tool(BT, "ListEventDispatchers", {"blueprint": bp}))

# 3. Broadcast it from the event graph. Read the grammar before the first Write, and read the
#    scope so a later Edit has a revision to work against.
vibeue.exec_tool(DSL, "Docs", {})
before = vibeue.exec_tool(DSL, "Read", {"assetPath": BPP, "scope": "EventGraph"})
rev = before["revision"]

res = vibeue.exec_tool(DSL, "Write", {"assetPath": BPP, "scope": "EventGraph", "dsl": """
(event `self.ReceiveAnyDamage`
  (`self.OnDied` (`self.GetInstigator`)))
"""})
rev = res["revision"]

# 4. Subscribe, in the SAME Blueprint for the sample: a handler whose parameters come from the
#    dispatcher's signature declares the source with :!signature, and |CreateDelegate wraps it.
res = vibeue.exec_tool(DSL, "Write", {"assetPath": BPP, "scope": "EventGraph", "dsl": """
(event `HandleDied`
  :!signature `self.OnDied`
  (params (Killer))
  (`KismetSystemLibrary.PrintString` "died"))

(event `self.ReceiveBeginPlay`
  (add `self.OnDied` (`|CreateDelegate` :!event `HandleDied`)))
"""})
rev = res["revision"]

# 5. Verify from the canonical text, not from what we sent. Write is UPSERT: the entries above
#    were added or replaced and every other entry was left alone (and warned about). Removing one
#    needs a tombstone: (event `Foo` :!delete).
after = vibeue.exec_tool(DSL, "Read", {"assetPath": BPP, "scope": "EventGraph"})
print(after["dsl"])
assert "OnDied" in after["dsl"], "broadcast/subscribe entries missing"

vibeue.exec_tool(BT, "CompileBlueprint", {"blueprint": bp})
vibeue.exec_tool(AS, "SaveAsset", {"assetPath": BP_PATH})
print("saved:", BP_PATH)
