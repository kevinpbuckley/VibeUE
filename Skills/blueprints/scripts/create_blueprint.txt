# create_blueprint.txt — Create a Blueprint with typed variables, set its defaults, compile, save.
#
# Sample script for the blueprints skill. Run via execute_python.
#
# UE 5.9: the Blueprint class itself is Epic's BlueprintToolsets.BlueprintToolset, and every
# property goes through EditorToolset.ObjectToolset. BlueprintService.create_blueprint /
# add_variable / add_member_variable / compile_blueprint / get_property / set_property were all
# cut; what survives there is Timelines, create_macro_graph, analyze_graph_layout and
# refresh_blueprint_editor.
#
# From an MCP client these are call_tool:
#   call_tool(tool_name="Create", toolset_name="BlueprintToolsets.BlueprintToolset",
#             arguments={"folderPath": FOLDER, "assetName": NAME,
#                        "assetType": {"refPath": "/Script/Engine.Character"}})
# Inside execute_python they are vibeue.exec_tool(toolset, tool, args), which fills missing
# optional args from the tool schema, unwraps "returnValue" and raises RuntimeError on error.
# Object/class arguments are {"refPath": "/Game/Dir/Asset.Asset"} handles either way — a bare
# package path with no ".Asset" suffix is rejected.
import json, unreal, vibeue

BT = "BlueprintToolsets.BlueprintToolset"
OT = "EditorToolset.ObjectToolset"
AS = "EditorToolset.AssetToolset"

NAME, FOLDER = "BP_Player", "/Game/Blueprints"
PARENT = {"refPath": "/Script/Engine.Character"}   # or /Script/Engine.Actor, .../Pawn, or a BP's _C
path = "{}/{}".format(FOLDER, NAME)

les = unreal.get_editor_subsystem(unreal.LevelEditorSubsystem)
assert not les.is_in_play_in_editor(), "stop PIE before editing a Blueprint"

if vibeue.exec_tool(AS, "Exists", {"path": path}):
    vibeue.exec_tool(AS, "Delete", {"path": path})

# 1. Create. folderPath and assetName are SEPARATE strings; the return is a {refPath} handle,
#    so reuse it instead of rebuilding one.
bp = vibeue.exec_tool(BT, "Create", {"folderPath": FOLDER, "assetName": NAME, "assetType": PARENT})
print("created:", bp)

# 2. Variables. AddVariable's string types: bool, int, float, byte, string, name, text, Vector,
#    Rotator, Transform, Vector2D, LinearColor, plus the bare name or object path of any UEnum or
#    UScriptStruct. Use int (NOT int32) and float (NOT double).
for name, type_name in (("Health", "float"), ("IsAlive", "bool")):
    vibeue.exec_tool(BT, "AddVariable", {"blueprint": bp, "name": name, "typeName": type_name})

# Container / category / tooltip / read-only all live in one options struct.
# NOTE: a MAP container uses FString as its key type.
vibeue.exec_tool(BT, "AddVariable", {"blueprint": bp, "name": "Waypoints", "typeName": "Vector",
                                     "options": {"containerType": "ARRAY", "category": "Patrol",
                                                 "tooltip": "Patrol route, world space"}})

# Anything the string list does not cover -> AddStructVariable / AddObjectVariable.
vibeue.exec_tool(BT, "AddObjectVariable",
                 {"blueprint": bp, "name": "DeathFX",
                  "objectClass": {"refPath": "/Script/Niagara.NiagaraSystem"}})

# 3. Compile — a variable only exists on the skeleton class after this.
vibeue.exec_tool(BT, "CompileBlueprint", {"blueprint": bp})

# 4. Defaults are CDO properties. There is no set_variable_default_value any more.
cdo = vibeue.exec_tool(BT, "GetDefaultObject", {"blueprint": bp})
vibeue.exec_tool(OT, "SetProperties", {"instance": cdo, "values": json.dumps(
    {"Health": 100.0, "IsAlive": True, "bReplicates": True})})
# String-keyed property access uses the NATIVE C++ name, so booleans keep the b prefix here
# ("bReplicates"), unlike Python attribute access where UE strips it.
print("defaults:", vibeue.exec_tool(OT, "GetProperties",
                                    {"instance": cdo, "properties": ["Health", "bReplicates"]}))

# 5. Verify, then save. ListVariables returns only name + writable; the type and default come
#    from ObjectToolset on the CDO.
print("variables:", vibeue.exec_tool(BT, "ListVariables", {"blueprint": bp}))
vibeue.exec_tool(BT, "CompileBlueprint", {"blueprint": bp})
vibeue.exec_tool(AS, "SaveAsset", {"assetPath": path})
print("saved:", path)
