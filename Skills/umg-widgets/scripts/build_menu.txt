# build_menu.txt — Build a vertical-menu hierarchy inside an existing Widget Blueprint.
#
# Sample script for the umg-widgets skill. Run via execute_python.
# Assumes WIDGET_PATH already exists (see create_widget.txt).
#
# TWO call styles in one script:
#   - the hierarchy is Epic's WidgetBlueprintEditorSubsystem -> vibeue.exec_tool(WS, "CreateAndAddWidget", ...)
#   - the label text is Epic's ObjectToolset -> vibeue.exec_tool("EditorToolset.ObjectToolset", ...)
#     (ObjectToolset has NO Python binding: unreal.ObjectToolset.set_properties() does not exist)
import json
import unreal
import vibeue

WIDGET_PATH = "/Game/Blueprints/TestWidget"
BUTTONS = [("PlayButton", "PLAY"), ("OptionsButton", "OPTIONS"), ("QuitButton", "QUIT")]

OT = "EditorToolset.ObjectToolset"
WS = "UMGEditor.WidgetBlueprintEditorSubsystem"
ASSET = WIDGET_PATH.rsplit("/", 1)[-1]
WBP_OBJ = f"{WIDGET_PATH}.{ASSET}"


def widget_obj(name):
    """Object path of a widget in the EDITABLE tree.

    Note `<Asset>:WidgetTree.<name>`, not `<Asset>_C:WidgetTree...` — the _C copy is rebuilt by
    compile, so writes there are silently discarded.
    """
    return f"{WIDGET_PATH}.{ASSET}:WidgetTree.{name}"


def add(widget_class, name, parent=None):
    args = {"widgetBlueprint": WBP_OBJ, "widgetClass": widget_class, "widgetDisplayName": name}
    if parent:
        args["parentWidget"] = widget_obj(parent)   # explicit parent; no parent + empty tree -> root
    return vibeue.exec_tool(WS, "CreateAndAddWidget", args)


# Root first (no parent on an empty tree -> becomes the root), then the container, then children.
add("/Script/UMG.CanvasPanel", "Root")
add("/Script/UMG.VerticalBox", "ButtonList", "Root")

for btn_name, label in BUTTONS:
    # Widget names are unique PER BLUEPRINT, not per parent — hence the per-button suffix.
    add("/Script/UMG.Button", btn_name, "ButtonList")
    text_name = f"{btn_name}Text"
    add("/Script/UMG.TextBlock", text_name, btn_name)
    vibeue.exec_tool(OT, "SetProperties", {
        "instance": widget_obj(text_name),
        "values": json.dumps({"Text": label}),   # `values` is a JSON *string*
    })

wbp = unreal.load_asset(WIDGET_PATH)
unreal.BlueprintToolset.compile_blueprint(wbp, False)
unreal.EditorAssetLibrary.save_asset(WIDGET_PATH)

# Verify before claiming success: DescribeWidget replaces every old VibeUE read.
res = unreal.WidgetStaticAnalysisToolset.describe_widget(wbp, None, -1, "", False)
print(res.available_widget_count, "widget(s)")
for e in res.widgets:
    parent = e.parent.get_name() if e.parent else "-"
    print(f"{'  ' * e.depth}{e.widget_name} ({e.widget_class_path}) parent={parent}")
