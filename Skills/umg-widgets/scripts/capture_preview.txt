# capture_preview.txt — Screenshot a Widget Blueprint's OPEN DESIGNER.
#
# Sample script for the umg-widgets skill. Run via execute_python.
#
# ### READ THIS BEFORE PROMISING A PREVIEW ###
# There is no longer a deterministic offscreen widget render.
#   * EditorAppToolset.CaptureAssetImage EXPLICITLY REFUSES Widget Blueprints. Its allowlist is
#     AnimationAsset, Skeleton, StaticMesh, SkeletalMesh, Texture, MaterialInterface; anything
#     else raises "Asset type does not support image capture".
#   * What survives is 3 calls, NEEDS THE DESIGNER OPEN, and is a SCREENSHOT OF THAT DESIGNER
#     cropped to a Slate widget — not an offscreen render at a size you choose. There is no
#     width/height argument. Tell the user that is what they are getting.
#
#       OpenEditorForAsset  ->  Snapshot  ->  Screenshot(ref)
#
# None of these toolsets has a Python binding (unreal.EditorAppToolset.* does not exist), so
# every call goes through vibeue.exec_tool — the in-process form of call_tool.
import vibeue

WIDGET_PATH = "/Game/Blueprints/TestWidget"
WIDGET_REF = ""     # paste a ref from the printed snapshot to crop to it; "" = the active window

EA = "EditorToolset.EditorAppToolset"
SI = "SlateInspectorToolset.SlateInspectorToolset"

# 1. The designer must be open — Screenshot fails on a widget in no visible window.
vibeue.exec_tool(EA, "OpenEditorForAsset", {"assetPath": WIDGET_PATH})

# 2. Snapshot the live Slate tree to find the designer's preview ref.
#    Each line carries [pos=X,Y size=W,H] in desktop pixels.
tree = vibeue.exec_tool(SI, "Snapshot", {"ref": "", "maxDepth": 30})
print(tree)
print("\n^ find the designer preview ref above, set WIDGET_REF, and re-run for a cropped shot.\n")

# 3. Crop to it. "" captures the active window (falling back to the main editor window).
#    Returns an FToolsetImage; through the MCP endpoint that is a real image content block.
img = vibeue.exec_tool(SI, "Screenshot", {"ref": WIDGET_REF})
print("captured:", type(img).__name__, (list(img.keys()) if isinstance(img, dict) else img))
