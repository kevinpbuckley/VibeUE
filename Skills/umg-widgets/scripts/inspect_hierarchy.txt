# inspect_hierarchy.txt — Print a Widget Blueprint's full hierarchy, slots, and properties.
#
# Sample script for the umg-widgets skill. Run via execute_python.
# ONE call does it all: WidgetStaticAnalysisToolset.DescribeWidget returns the walk-order tree,
# each widget's Properties[], and its SlotProperties[]. It replaces every removed VibeUE read.
# DescribeWidget is PythonCallable, so unreal.WidgetStaticAnalysisToolset.* works directly.
import unreal

WIDGET_PATH = "/Game/Blueprints/TestWidget"
SKIP_DEFAULTS = False   # True (the tool's default) hides every value equal to the class default
PROPERTY_FILTER = ""    # substring match on property name; "" = every property
MAX_DEPTH = -1          # -1 = no limit; 0 = StartWidget only; N = N levels

wbp = unreal.load_asset(WIDGET_PATH)
if wbp is None:
    raise RuntimeError(f"no Widget Blueprint at {WIDGET_PATH}")

res = unreal.WidgetStaticAnalysisToolset.describe_widget(
    wbp,
    None,              # StartWidget: None = the whole tree
    MAX_DEPTH,
    PROPERTY_FILTER,
    SKIP_DEFAULTS,
    False,             # bIncludeFullDescription: True also fills res.description (verbose text)
    False,             # bIncludeComposedColors: True fills each entry's composed_colors
)

# available_widget_count is the canary: if it exceeds len(res.widgets), the list got cut.
print(f"{res.available_widget_count} widget(s) in {WIDGET_PATH}; {len(res.widgets)} returned")

for e in res.widgets:
    pad = "  " * e.depth
    if e.parent:
        where = f"parent={e.parent.get_name()}"
    elif e.named_slot_host:
        where = f"named-slot on {e.named_slot_host.get_name()}"
    else:
        where = "ROOT (or inherited named slot)"
    flags = " ".join(f for f, on in (("var", e.is_variable), ("inherited", e.inherited)) if on)
    print(f"{pad}- {e.widget_name} ({e.widget_class_path}) {where} {flags}")
    for p in e.properties:
        print(f"{pad}    {p.property_name} = {p.exported_value}")
    for p in e.slot_properties:
        print(f"{pad}    [slot] {p.property_name} = {p.exported_value}")

print("tree depth:", unreal.WidgetStaticAnalysisToolset.get_widget_tree_depth(wbp))
