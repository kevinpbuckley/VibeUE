# apply_font.txt — Apply a full font configuration to a text widget and verify it.
#
# Sample script for the umg-widgets skill. Run via execute_python.
# FSlateFontInfo is a plain USTRUCT(BlueprintType), so a font is an ORDINARY property write
# through Epic's ObjectToolset — there is no dedicated font API any more.
# ObjectToolset has NO Python binding, so every call here goes through vibeue.exec_tool
# (the in-process form of call_tool).
import json
import unreal
import vibeue

WIDGET_PATH = "/Game/Blueprints/TestWidget"
TEXT_WIDGET = "HeaderTitle"   # must be a TextBlock that already exists

OT = "EditorToolset.ObjectToolset"
ASSET = WIDGET_PATH.rsplit("/", 1)[-1]
TXT = f"{WIDGET_PATH}.{ASSET}:WidgetTree.{TEXT_WIDGET}"   # the EDITABLE tree, not <Asset>_C

# 1. Read the real struct shape BEFORE composing JSON. Member names are FSlateFontInfo's.
print("--- Font schema ---")
print(vibeue.exec_tool(OT, "GetPropertySchema", {"instance": TXT, "root": "Font", "maxDepth": 1}))

# 2. Write it. `values` is a JSON *string*, so json.dumps().
#    ColorAndOpacity on a TextBlock is an FSlateColor (SpecifiedColor + ColorUseRule);
#    ShadowColorAndOpacity is a plain FLinearColor. UMG colours are LINEAR.
vibeue.exec_tool(OT, "SetProperties", {"instance": TXT, "values": json.dumps({
    "Font": {
        "FontObject": "/Engine/EngineFonts/Roboto.Roboto",
        "TypefaceFontName": "Bold",
        "Size": 30.0,
        "LetterSpacing": 20,
    },
    "ColorAndOpacity": {
        "SpecifiedColor": {"R": 1.0, "G": 0.9, "B": 0.25, "A": 1.0},   # yellow, linear
        "ColorUseRule": "UseColor_Specified",
    },
    "ShadowOffset": {"X": 2.0, "Y": 2.0},
    "ShadowColorAndOpacity": {"R": 0.0, "G": 0.0, "B": 0.0, "A": 0.75},
})})

wbp = unreal.load_asset(WIDGET_PATH)
unreal.BlueprintToolset.compile_blueprint(wbp, False)
unreal.EditorAssetLibrary.save_asset(WIDGET_PATH)

# 3. Read back every field that was set — SetProperties returning True is not proof of the value.
print("--- readback ---")
print(vibeue.exec_tool(OT, "GetProperties", {"instance": TXT, "properties": [
    "Font.TypefaceFontName", "Font.Size", "Font.LetterSpacing",
    "ColorAndOpacity", "ShadowOffset", "ShadowColorAndOpacity",
]}))
