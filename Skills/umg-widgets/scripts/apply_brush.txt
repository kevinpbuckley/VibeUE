# apply_brush.txt — Apply a full brush configuration to an image widget and verify it.
#
# Sample script for the umg-widgets skill. Run via execute_python.
# FSlateBrush is a plain USTRUCT(BlueprintType), so a brush is an ORDINARY property write through
# Epic's ObjectToolset — there is no dedicated brush API, and no slot_name argument, any more.
#
# The brush PROPERTY NAME differs per widget: "Brush" on an Image, "Background" on a Border,
# "WidgetStyle.Normal"/"Hovered"/"Pressed" on a Button. ListProperties finds the real one.
import json
import unreal
import vibeue

WIDGET_PATH = "/Game/Blueprints/TestWidget"
IMAGE_WIDGET = "BackgroundImage"   # must be an Image that already exists
BRUSH_PROPERTY = "Brush"           # an Image's brush property

OT = "EditorToolset.ObjectToolset"
ASSET = WIDGET_PATH.rsplit("/", 1)[-1]
IMG = f"{WIDGET_PATH}.{ASSET}:WidgetTree.{IMAGE_WIDGET}"   # the EDITABLE tree, not <Asset>_C

# 1. Find the brush property and read its real shape before composing JSON.
print("--- brush-ish properties ---")
print(vibeue.exec_tool(OT, "ListProperties",
                       {"instance": IMG, "nameFilter": "brush|background|style", "maxDepth": 1}))
print("--- schema ---")
print(vibeue.exec_tool(OT, "GetPropertySchema",
                       {"instance": IMG, "root": BRUSH_PROPERTY, "maxDepth": 2}))

# 2. Write it. TintColor is an FSlateColor (SpecifiedColor + ColorUseRule); rounded-box corner
#    radii live under OutlineSettings.CornerRadii (a Vector4), not a top-level CornerRadius.
vibeue.exec_tool(OT, "SetProperties", {"instance": IMG, "values": json.dumps({
    BRUSH_PROPERTY: {
        "ResourceObject": "/Engine/EngineResources/WhiteSquareTexture.WhiteSquareTexture",
        "DrawAs": "RoundedBox",
        "TintColor": {
            "SpecifiedColor": {"R": 0.08, "G": 0.12, "B": 0.2, "A": 1.0},   # dark blue, linear
            "ColorUseRule": "UseColor_Specified",
        },
        "ImageSize": {"X": 1920.0, "Y": 1080.0},
        "Margin": {"Left": 0.2, "Top": 0.2, "Right": 0.2, "Bottom": 0.2},
        "OutlineSettings": {"CornerRadii": {"X": 16.0, "Y": 16.0, "Z": 16.0, "W": 16.0}},
    },
})})

wbp = unreal.load_asset(WIDGET_PATH)
unreal.BlueprintToolset.compile_blueprint(wbp, False)
unreal.EditorAssetLibrary.save_asset(WIDGET_PATH)

# 3. Read back to verify.
print("--- readback ---")
print(vibeue.exec_tool(OT, "GetProperties", {"instance": IMG, "properties": [
    f"{BRUSH_PROPERTY}.DrawAs",
    f"{BRUSH_PROPERTY}.TintColor",
    f"{BRUSH_PROPERTY}.Margin",
    f"{BRUSH_PROPERTY}.OutlineSettings.CornerRadii",
]}))

# DrawAs values: Image, Box, Border, RoundedBox, NoDrawType.

# bIncludeComposedColors=True and read each entry's composed_colors (linear, per interact state).
