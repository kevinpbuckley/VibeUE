# inspect_sequence.txt — Safely inspect AnimSequences for a skeleton (metadata, no asset loading).
#
# Sample script for the animsequence skill. Run via execute_python.
# Use the service discovery methods — NEVER unreal.load_asset() in a loop (memory crash).
import unreal
ass = unreal.AnimSequenceService

# Pick a skeleton out of animation metadata. VibeUE's own skeleton list call is gone; to search for
# skeleton assets directly, use Epic's EditorToolset.AssetToolset FindAssets filtered to class
# Skeleton (async — see the SAFE DISCOVERY box in SKILL.md).
skeletons = [a.skeleton_path for a in ass.list_anim_sequences("/Game", "") if a and a.skeleton_path]
if not skeletons:
    print("no skeletons found");
else:
    skeleton_path = skeletons[0]
    anims = ass.find_animations_for_skeleton(skeleton_path)
    print(f"{len(anims)} animations for {skeleton_path}")
    for a in anims[:5]:
        info = ass.get_anim_sequence_info(a.anim_path)
        print(f"  {a.anim_name}: frames={a.frame_count} dur={a.duration:.2f}s")
