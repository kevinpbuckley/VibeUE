# add_socket.txt — Add a socket to a skeletal mesh and verify (ENGINE SkeletalMeshToolset).
#
# Sample script for the skeleton skill. Run it as the script of one execute_python call.
#
# ARCHITECTURE: socket CRUD and bone enumeration are owned by the ENGINE toolset
# EditorToolset.SkeletalMeshToolset. Inside execute_python, call its tools with the injected
# `await call_tool("<toolset>.<Tool>", **camelCaseArgs)`, which returns the tool's value directly.
# VibeUE's SkeletonService keeps bone transform/hierarchy edits, retargeting, curves, blend
# profiles, and constraints (not sockets). Check a tool's parameters with
# get_symbol("EditorToolset.SkeletalMeshToolset.SetSocketTransform").
import unreal

TS = "EditorToolset.SkeletalMeshToolset"
MESH = unreal.load_asset("/Game/Characters/SKM_Mannequin")   # pass Unreal objects as they are
SOCKET = "WeaponSocket"
BONE = "hand_r"

# Add the socket (always attached to the skeletal mesh; there is no add-to-skeleton flag):
await call_tool(TS + ".AddSocket", mesh=MESH, socketName=SOCKET, boneName=BONE)

# Set its offset relative to the bone. translation / rotation / scale are each optional:
await call_tool(TS + ".SetSocketTransform", mesh=MESH, socketName=SOCKET,
                translation=unreal.Vector(10, 0, 0),
                rotation=unreal.Rotator(roll=0, pitch=0, yaw=90))

# Verify: the final expression is what execute_python returns.
{"sockets": await call_tool(TS + ".GetSocketNames", mesh=MESH),
 "bone": await call_tool(TS + ".GetSocketBone", mesh=MESH, socketName=SOCKET)}
